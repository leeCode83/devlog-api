pydantic-settings       # Environment variable management for Pydantic

redis[asyncio]          # Redis async client (analytics cache + rate limiting)

# ── Logging ──────────────────────────────────────────────────
structlog                # Structured JSON logging (production-grade)

# ── Testing ──────────────────────────────────────────────────
pytest                   # Test framework
pytest-asyncio           # Async test support for pytest

# ── Utilities ───────────────────────────────────────────────
python-multipart          # Form data parsing (for file uploads in APIs)
